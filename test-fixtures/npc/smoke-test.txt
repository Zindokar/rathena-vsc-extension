prontera,150,180,4	script	Zindokar Test	100,{

	mes "[^0055FFZindokar^000000]";
	mes "Hola, " + strcharinfo(0) + ". Bienvenido.";
	next;

	// Los nueve scopes de variable, cada uno de un color
	.@precio = 5000;              // scope local del script
	.npc_contador++;              // del NPC
	@temporal = 1;                // temporal del personaje
	permanente_pj = 2;            // permanente del personaje
	#puntos_cuenta = 10;          // cuenta, este map-server
	##puntos_globales = 20;       // cuenta, login-server
	$global_permanente = 30;      // global permanente
	$@global_temporal = 40;       // global temporal
	'variable_instancia = 50;     // de instancia
	.@nombre$ = "soy una cadena"; // sufijo $ = tipo cadena

	// Parametros integrados del personaje
	if (Zeny < .@precio || BaseLevel < 50) {
		mes "No cumples los requisitos.";
		close;
	}
	Zeny -= .@precio;

	switch (select("Curarme:Darme un item:Salir")) {
		case 1:
			percentheal 100,100;
			sc_start SC_INCREASEAGI,240000,10;
			specialeffect2 EF_HEAL2;
			break;
		case 2:
			getitem 501,10;
			getitem "Apple",1;
			break;
		case 3:
			close;
	}

	for (.@i = 0; .@i < 3; .@i++) {
		dispbottom "Vuelta " + .@i;
	}

	// Numeros hexadecimales y operadores largos
	.@mascara = 0xFF00;
	.@mascara <<= 2;

	// Constante en minusculas: la busqueda es case-insensitive
	if (Class == Job_Novice) {
		mes "Eres un novato.";
	}

	end;

OnInit:
	.npc_contador = 0;
	end;

OnPCLoginEvent:
	dispbottom "Bienvenido al servidor.";
	end;

OnTouch:
	end;

L_EtiquetaPropia:
	end;
}
